import java.util.*;
import steam_boiler.*;
import steam_boiler.domain.*;
import steam_boiler.event.*;

public class Sim {
    double level; double steam; boolean[] on = new boolean[4]; boolean[] onBefore = new boolean[4];
    boolean valve; boolean waiting = true; boolean ready = false; boolean programReady = false;
    int cycle = 0; double steamTarget = 8.0; boolean steamStarted = false;
    boolean levelBroken = false; boolean sendLevelRepair = false;
    List<InputEvent> extra = new ArrayList<>();
    SteamBoilerProgram p = new SteamBoilerProgram();

    List<OutputEvent> run() {
        List<InputEvent> in = new ArrayList<>();
        if (waiting) in.add(new InputEvent.SteamBoilerWaiting());
        if (programReady && p.currentMode() == OperatingMode.INITIALIZATION) in.add(new InputEvent.PhysicalUnitsReady());
        in.add(new InputEvent.Level(levelBroken ? 2000.0 : level));
        in.add(new InputEvent.Steam(steam));
        for (int n = 0; n < 4; n++) {
            in.add(new InputEvent.PumpState(new PumpReport(n, on[n])));
            boolean flow = on[n] && onBefore[n];
            in.add(new InputEvent.PumpControlState(new FlowReport(n, flow ? WaterFlow.WATER_FLOWING : WaterFlow.WATER_NOT_FLOWING)));
        }
        in.addAll(extra); extra.clear();
        List<OutputEvent> out = p.runCycle(in);
        boolean[] prevOn = on.clone();
        for (OutputEvent o : out) {
            if (o instanceof OutputEvent.OpenPump) on[((OutputEvent.OpenPump) o).pumpNumber()] = true;
            if (o instanceof OutputEvent.ClosePump) on[((OutputEvent.ClosePump) o).pumpNumber()] = false;
            if (o instanceof OutputEvent.Valve) valve = !valve;
            if (o instanceof OutputEvent.ProgramReady) programReady = true;
            if (o instanceof OutputEvent.LevelFailureDetection) extra.add(new InputEvent.LevelFailureAcknowledgement());
        }
        // physics over the next 5 s
        int pouring = 0;
        for (int n = 0; n < 4; n++) { if (on[n] && prevOn[n]) pouring++; }
        double oldSteam = steam;
        if (steamStarted && steam < steamTarget) steam = Math.min(steamTarget, steam + 4.0);
        double steamOut = 5.0 * (oldSteam + steam) / 2.0;
        level = level + pouring * 5.0 * 5.0 - steamOut - (valve ? 50.0 : 0.0);
        onBefore = prevOn;   // state reported next cycle: on now, and was on before
        for (int n = 0; n < 4; n++) onBefore[n] = prevOn[n] && on[n];
        cycle++;
        return out;
    }

    static String brief(List<OutputEvent> out) {
        StringBuilder sb = new StringBuilder();
        for (OutputEvent o : out) sb.append(o.getClass().getSimpleName()).append(o instanceof OutputEvent.OpenPump || o instanceof OutputEvent.ClosePump ? "" : "").append(' ');
        return sb.toString();
    }

    public static void main(String[] a) {
        for (double start : new double[] {300.0, 500.0, 700.0}) {
            Sim s = new Sim(); s.level = start;
            OperatingMode last = null; int modeChanges = 0;
            for (int i = 0; i < 120; i++) {
                if (s.p.currentMode() == OperatingMode.NORMAL) s.steamStarted = true;
                if (i == 60) s.levelBroken = true;            // level device fails
                if (i == 63) { s.levelBroken = false; s.extra.add(new InputEvent.LevelRepaired()); }
                double lv = s.level;
                List<OutputEvent> out = s.run();
                OperatingMode m = s.p.currentMode();
                if (m != last) { System.out.printf("start=%5.0f cycle %3d level %6.1f steam %4.1f -> %s  [%s]%n", start, i, lv, s.steam, m, brief(out)); last = m; }
                if (s.p.hasStopped()) break;
            }
            System.out.printf("start=%5.0f final mode %s level %.1f%n%n", start, s.p.currentMode(), s.level);
        }
        // STOP three times
        Sim s = new Sim(); s.level = 500.0;
        for (int i = 0; i < 6; i++) { s.run(); }
        for (int i = 0; i < 3; i++) { s.extra.add(new InputEvent.StopRequest()); s.run(); System.out.println("after STOP " + (i+1) + ": " + s.p.currentMode()); }
        // missing message => transmission failure
        Sim t = new Sim(); t.level = 500.0;
        for (int i = 0; i < 6; i++) { t.run(); }
        List<InputEvent> partial = new ArrayList<>(); partial.add(new InputEvent.Level(500.0));
        t.p.runCycle(partial); System.out.println("after missing data: " + t.p.currentMode());
    }
}
