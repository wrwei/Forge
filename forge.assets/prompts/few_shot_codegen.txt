FEW-SHOT EXAMPLE: MODE-NESTED IF-ELSE STATE MACHINE

This example shows the required code structure for a small traffic-light
controller with three modes (RED, GREEN, YELLOW) and named boolean predicates.
The domain is fictional; adapt the pattern to the actual requirements. Every
rule it follows is stated in `java_codegen_rules.txt`.

// FILE: src/main/java/com/example/controller/annotation/RoboChartType.java
package com.example.controller.annotation;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Retention(RetentionPolicy.SOURCE)
@Target({ElementType.FIELD, ElementType.PARAMETER, ElementType.METHOD})
public @interface RoboChartType {
    String value();
}

// FILE: src/main/java/com/example/controller/event/InputEvent.java
package com.example.controller.event;

import com.example.controller.annotation.RoboChartType;

// Sealed interface: one record per event type.
// Signal events carry no data; data-carrying events hold a payload.
public sealed interface InputEvent {
    record TimerExpired() implements InputEvent {}
    record EmergencyStop() implements InputEvent {}
    record Resume() implements InputEvent {}
    record SetSpeed(@RoboChartType("real") double value) implements InputEvent {}
}

// FILE: src/main/java/com/example/controller/mode/TrafficMode.java
package com.example.controller.mode;

// One value per operating mode; the initial mode first.
public enum TrafficMode {
    RED,
    GREEN,
    YELLOW
}

// FILE: src/main/java/com/example/controller/mode/Signal.java
package com.example.controller.mode;

// The values the actuator can show (an enum, not a String).
public enum Signal {
    RED,
    GREEN,
    YELLOW
}

// FILE: src/main/java/com/example/controller/TrafficConstants.java
package com.example.controller;

import com.example.controller.annotation.RoboChartType;

public final class TrafficConstants {
    @RoboChartType("nat") public static final int LONG_QUEUE = 10;
    @RoboChartType("real") public static final double MAX_WAIT = 30.0;
    private TrafficConstants() {}
}

// FILE: src/main/java/com/example/controller/sensor/Sensor.java
package com.example.controller.sensor;

import com.example.controller.annotation.RoboChartType;

// Sensor class: quantities read from, or derived from, the environment.
public final class Sensor {

    @RoboChartType("nat") private int vehicleCount;
    @RoboChartType("real") private double waitTime;

    public void update(@RoboChartType("nat") int vehicleCount, @RoboChartType("real") double waitTime) {
        this.vehicleCount = vehicleCount;
        this.waitTime = waitTime;
    }

    @RoboChartType("nat") public int vehicleCount() { return vehicleCount; }
    @RoboChartType("real") public double waitTime() { return waitTime; }
}

// FILE: src/main/java/com/example/controller/actuator/Actuator.java
package com.example.controller.actuator;

import com.example.controller.mode.Signal;

// Actuator: receives output commands and records the last one.
public final class Actuator {
    private Signal lastSignal = Signal.RED;

    public void setSignal(Signal signal) { this.lastSignal = signal; }
    public Signal lastSignal() { return lastSignal; }
}

// FILE: src/main/java/com/example/controller/TrafficController.java
package com.example.controller;

import com.example.controller.actuator.Actuator;
import com.example.controller.event.InputEvent;
import com.example.controller.mode.Signal;
import com.example.controller.mode.TrafficMode;
import com.example.controller.sensor.Sensor;

public final class TrafficController {
    private TrafficMode currentMode = TrafficMode.RED;
    private final Sensor sensor;
    private final Actuator actuator;

    public TrafficController(Sensor sensor, Actuator actuator) {
        this.sensor = sensor;
        this.actuator = actuator;
    }

    public TrafficMode currentMode() { return currentMode; }

    public void step(InputEvent event) {
        // --- Named boolean predicates (declared BEFORE the if-else chain) ---
        boolean queueLong = sensor.vehicleCount() > TrafficConstants.LONG_QUEUE;
        boolean waitAboveLimit = sensor.waitTime() > TrafficConstants.MAX_WAIT;

        // --- Two-level if-else: every outer branch is currentMode == X ---
        if (currentMode == TrafficMode.RED) {
            if (event instanceof InputEvent.TimerExpired && queueLong) {
                currentMode = TrafficMode.GREEN;
                actuator.setSignal(Signal.GREEN);
            } else if (event instanceof InputEvent.TimerExpired && waitAboveLimit) {
                currentMode = TrafficMode.GREEN;
                actuator.setSignal(Signal.GREEN);
            }

        } else if (currentMode == TrafficMode.GREEN) {
            // An override that applies in several modes comes first in each.
            if (event instanceof InputEvent.EmergencyStop) {
                currentMode = TrafficMode.RED;
                actuator.setSignal(Signal.RED);
            } else if (event instanceof InputEvent.TimerExpired) {
                currentMode = TrafficMode.YELLOW;
                actuator.setSignal(Signal.YELLOW);
            }
            // Autonomous (guard-only) transition: no event test.
            else if (!queueLong && !waitAboveLimit) {
                currentMode = TrafficMode.YELLOW;
                actuator.setSignal(Signal.YELLOW);
            }

        } else if (currentMode == TrafficMode.YELLOW) {
            if (event instanceof InputEvent.EmergencyStop) {
                currentMode = TrafficMode.RED;
                actuator.setSignal(Signal.RED);
            } else if (event instanceof InputEvent.TimerExpired) {
                currentMode = TrafficMode.RED;
                actuator.setSignal(Signal.RED);
            }
        }
    }
}
